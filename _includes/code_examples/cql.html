-- schema.cql — KillrVideo Cassandra Schema -- Designed for query-first data modeling CREATE KEYSPACE killrvideo WITH replication = { 'class': 'NetworkTopologyStrategy', 'datacenter1': 3 }; -- Videos by user (supports "my videos" feed) CREATE TABLE killrvideo.videos_by_user ( user_id UUID, added_date TIMESTAMP, video_id UUID, title TEXT, tags SET<TEXT>, PRIMARY KEY ((user_id), added_date, video_id) ) WITH CLUSTERING ORDER BY (added_date DESC); -- SAI index for tag-based search CREATE CUSTOM INDEX videos_tags_idx ON killrvideo.videos_by_user (tags) USING 'StorageAttachedIndex'; -- Distributed counter table for ratings CREATE TABLE killrvideo.video_ratings ( video_id UUID PRIMARY KEY, rating_one COUNTER, rating_five COUNTER );
