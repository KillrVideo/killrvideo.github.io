// VideoService.java — KillrVideo Video Service // Java SDK with DataStax Astra DB client import com.datastax.astra.client.DataAPIClient; import com.datastax.astra.client.Collection; import com.datastax.astra.client.model.Document; import com.datastax.astra.client.model.FindOptions; public class VideoService { private final Collection<Document> videos; public VideoService(DataAPIClient client) { this.videos = client .getDatabase(System.getenv("ASTRA_DB_API_ENDPOINT")) .getCollection("videos"); } public String submitVideo(String userId, String title, List<String> tags) { Document doc = new Document() .id(UUID.randomUUID()) .append("userId", userId) .append("title", title) .append("tags", tags) .append("addedDate", Instant.now()); return videos.insertOne(doc) .getInsertedId() .toString(); } public List<Document> searchVideos(float[] embedding, int limit) { return videos.find(new FindOptions() .sort(Sorts.vector(embedding)) .limit(limit)) .all(); } }
