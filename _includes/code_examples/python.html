# video_service.py — KillrVideo Video Service # Demonstrates: partition key design + vector search from astrapy import DataAPIClient from datetime import datetime, UTC from uuid import uuid4 class VideoService: def __init__(self, db): self.videos = db.get_collection("videos") async def submit_video(self, user_id: str, title: str, description: str, tags: list[str]): """Insert a new video. user_id is the partition key.""" video_id = str(uuid4()) await self.videos.insert_one({ "_id": video_id, "userId": user_id, "title": title, "tags": tags, "addedDate": datetime.now(UTC).isoformat(), }) return video_id async def search_videos(self, query_embedding: list[float], limit=20): """Semantic search using vector similarity.""" cursor = self.videos.find( {}, sort={"$vector": query_embedding}, limit=limit, include_similarity=True ) return await cursor.to_list()
