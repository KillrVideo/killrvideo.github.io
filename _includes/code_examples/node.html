// videoService.js — KillrVideo Video Service // Node.js + @datastax/astra-db-ts SDK const { AstraDB } = require("@datastax/astra-db-ts"); class VideoService { constructor(token, apiEndpoint) { const client = new AstraDB(token, apiEndpoint); this.videos = client.collection("videos"); } async submitVideo({ userId, title, description, tags }) { const videoId = crypto.randomUUID(); await this.videos.insertOne({ _id: videoId, userId, title, description, tags, addedDate: new Date().toISOString(), }); return videoId; } async searchVideos(queryEmbedding, limit = 20) { return await this.videos.find({}, { sort: { $vector: queryEmbedding }, limit, includeSimilarity: true, }).toArray(); } }
